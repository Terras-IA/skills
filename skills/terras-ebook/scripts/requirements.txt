# --- Pipeline de vídeo (rotas A e B) ----------------------------------------
google-api-python-client   # busca dos vídeos na YouTube Data API v3
python-dotenv              # leitura do .env
requests                   # HTTP: Supadata e checagem de Shorts
youtube-transcript-api     # transcrição grátis (a Supadata é o fallback pago)
ebooklib                   # montagem do EPUB
markdown                   # markdown -> HTML dos capítulos

# --- Rota PDF (montagem do livro e verificação) ------------------------------
reportlab                  # desenha capa, miolo, folio e fio-de-pé
pypdf                      # merge da capa com o miolo, links e marcadores
pdfplumber                 # mede as caixas do sumário e acha a página de cada capítulo
fonttools                  # converte a fonte variável da marca (woff2) em TTF
brotli                     # necessário para o fonttools ler woff2
pymupdf                    # conferir links e texto extraído (o pypdf não resolve
                           # destino em número de página; o PyMuPDF resolve)
